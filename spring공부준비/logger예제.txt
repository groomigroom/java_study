import org.springframework.stereotype.Service;

@Slf4j  // 로거 객체를 자동으로 생성해 줍니다.
@Service
public class MemberService {

    public void registerMember(String name) {
        log.info("회원 가입 시작: {}", name); // 중요한 정보 기록

        if (name == null || name.trim().isEmpty()) {
            log.warn("경고: 회원 이름이 비어 있습니다."); // 잠재적 문제
            return;
        }

        try {
            // 비즈니스 로직 처리 예시
            log.debug("디버그 모드: 이름 길이 = {}", name.length()); // 상세 개발 정보
            log.info("회원 가입 완료: {}", name);
            
        } catch (Exception e) {
            log.error("에러 발생: 회원 가입 실패 - {}", e.getMessage()); // 심각한 오류
        }
    }
}

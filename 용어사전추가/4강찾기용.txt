@Transactional(readOnly = true)


@Configuration
@EnableWebSecurity


UserEntity entity = UserEntity.builder()
            .username(dto.getUsername())
            .password(passwordEncoder.encode(dto.getPassword()))
            .isLock(false)
            .isSocial(false)
            .roleType(UserRoleType.USER) // 우선 일반 유저로 가입
            .nickname(dto.getNickname())
            .email(dto.getEmail())
            .build();

List나 String[] 배열 형태로 명확히 생성해야 합니다. 2. 타임리프(Thymeleaf) Property 객체 접근 오류 * 현재 모델에 담아 보낸 데이터는 단순 문자열(String)들의 묶음입니다. * 하지만 HTML 코드에서는 th:text="${user.name}" 처럼 user 객체 내부에 name이라는 멤버 변수(Getter)가 있는 것처럼 접근하고 있어 Property 또는 필드를 찾을 수 없다는 에러(500)가 발생합니다. 단순 문자열을 출력할 때는 ${user}만 적어야 합니다. ------------------------------ ## 🛠️ 올바르게 수정한 코드## 1. 자바 스프링 컨트롤러 수정 중괄호 대입 대신 자바의 List.of() 또는 Arrays.asList()를 사용하여 단순 문자열 리스트를 생성하도록 변경합니다. [2] ## 1) 컨트롤러 파일 수정 (HelloController.java) package groomi.groomgroom; import org.springframework.stereotype.Controller;import org.springframework.ui.Model;import org.springframework.web.bind.annotation.GetMapping;import java.util.List; @Controllerpublic class HelloController { // 1. name 속성을 가질 수 있는 데이터를 레코드로 정의합니다. (클래스 내부나 외부에 선언 가능) public record UserDto(String name) {} @GetMapping("hello") public String hello(Model model) { // 2. 생성한 객체(UserDto)들을 List에 담아 모델에 추가합니다. List userList = List.of( new UserDto("김구름"), new UserDto("멍멍이"), new UserDto("비숑") ); model.addAttribute("users", userList); return "hello"; } } ------------------------------ ## 방법 2. 일반 Class로 구현하기 (Lombok 활용 혹은 Getter 필수) 전통적인 방식으로 클래스를 만드는 경우 타임리프가 내부적으로 데이터에 접근하기 위해서 Getter 메서드가 반드시 존재해야만 ${user.name} 문법이 작동합니다. [3] ## 1) 클래스 정의 (User.java) HelloController와 동일한 패키지 내부에 새로운 자바 클래스 파일을 생성합니다. package groomi.groomgroom; public class User { private String name; public User(String name) { this.name = name; } // ⭐ 타임리프의 ${user.name}은 이 getName() 메서드를 내부적으로 호출합니다! public String getName() { return name; } } ## 2) 컨트롤러 수정 (HelloController.java) package groomi.groomgroom; import org.springframework.stereotype.Controller;import org.springframework.ui.Model;import org.springframework.web.bind.annotation.GetMapping;import java.util.List; @Controllerpublic class HelloController { @GetMapping("hello") public String hello(Model model) { // 정의한 User 객체를 생성하여 리스트로 묶어 모델에 보냅니다. List userList = List.of( new User("김구름"), new User("멍멍이"), new User("비숑") ); model.addAttribute("users", userList); return "hello"; } } ------------------------------ ## 2. 타임리프 HTML 파일은 유지 (hello.html) 위 두 방법 중 하나로 자바 코드를 교체하고 나면, 질문해 주신 기존 HTML 코드가 수정 없이 정상 작동하게 됩니다.

public class HelloController {

    @GetMapping("hello")
    public String hello (Model model) {
        model.addAttribute(attributeName: "data", attributeValue: "hello!!");
        return "hello";
    }

    @GetMapping("hello_string")
    @ResponseBody
    public String helloString(@RequestParam("name") String name) {
        return "hello " + name;
    }
